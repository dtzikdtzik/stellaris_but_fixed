	planet_limit = 1
	empire_limit = 1
	icon = building_production_center

	category = manufacturing

		industrial
		factory
		foundry
	}

	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
		}
		NOR = {
			has_modifier = resort_colony
			has_modifier = slave_colony
		}
	}

	destroy_trigger = {
		exists = owner

			owner = { is_regular_empire = no }

			has_modifier = slave_colony
		}
	}

	allow = {
		has_major_upgraded_capital = yes
	}

	inline_script = {
		script = "jobs/politician_add"
		AMOUNT = @building_static_jobs
	}

#		planet_artisans_produces_mult = 0.15
#	}

	country_modifier = {
		country_consumer_goods_produces_mult = 0.05
		country_alloys_produces_mult = 0.05
	}

	resources = {
		category = planet_buildings
		cost = {
			minerals = @b2_minerals
			volatile_motes = @b2_rare_cost
		}
		upkeep = {
			energy = @b2_upkeep
			volatile_motes = @b2_rare_upkeep
		}
	}

	prerequisites = {
		"tech_global_production_strategy"
	}

	show_tech_unlock_if = {
		is_regular_empire = yes
	}

	inline_script = {
		script = districts/ai_alloys_extra_weighting
		AMOUNT = 10
	}

	inline_script = {
		script = districts/ai_consumer_goods_extra_weighting
		AMOUNT = 10
	}
