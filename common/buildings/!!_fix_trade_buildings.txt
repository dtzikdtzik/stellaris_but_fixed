	base_buildtime = @b2_time
	planet_limit = 1
	category = trade

	building_sets = {
		trade
	}

	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
		}
		NOR = {
			has_modifier = slave_colony
			has_modifier = penal_colony
		}
	}

	destroy_trigger = {
		exists = owner
		OR = {
			owner = { is_regular_empire = no }
			has_modifier = slave_colony
			has_modifier = penal_colony
		}
	}

	allow = {
	}

	inline_script = {
		script = jobs/trader_add

		}

	}
